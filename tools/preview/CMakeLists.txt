cmake_minimum_required(VERSION 3.16)
project(pet_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
option(PET_BUILD_DESKTOP "Build the native desktop framebuffer library" OFF)
if(PET_BUILD_DESKTOP)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
endif()
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
set(PET_UI_SOURCES
    ../../main/model_picker.cpp
    ../../main/thinker_transition.cpp
    ../../main/generated/thinker_camera_assets.cpp
    ../../main/theme_character.cpp ../../main/character_motion.cpp ../../main/character_cache.cpp
    ../../main/generated/character_assets.cpp ../../main/generated/vangogh_scene.cpp
    ../../main/font_cn_18.c ../../main/font_cn_30.c ../../main/font_menu_18.c
    ../../main/font_pixel_16.c ../../main/font_pixel_24.c)
add_executable(pet_preview renderer.cpp ${PET_UI_SOURCES})
target_compile_definitions(pet_preview PRIVATE PET_DESKTOP_PREVIEW=1)
target_include_directories(pet_preview PRIVATE ../../main)
target_link_libraries(pet_preview PRIVATE lvgl m)
enable_testing()
add_executable(astra_timing_test astra_timing_test.cpp)
target_include_directories(astra_timing_test PRIVATE ../../main)
target_compile_options(astra_timing_test PRIVATE -UNDEBUG)
add_test(NAME astra_local_period COMMAND astra_timing_test)
add_executable(model_button_test model_button_test.cpp)
target_include_directories(model_button_test PRIVATE ../../main)
target_compile_options(model_button_test PRIVATE -UNDEBUG)
add_test(NAME physical_model_button COMMAND model_button_test)
add_test(NAME ui_contracts COMMAND pet_preview --test)
add_test(NAME model_picker_preview COMMAND pet_preview --picker ${CMAKE_CURRENT_BINARY_DIR}/model-picker)
add_executable(motion_test motion_test.cpp ../../main/character_motion.cpp)
target_include_directories(motion_test PRIVATE ../../main)
add_test(NAME character_motion COMMAND motion_test)
add_executable(model_test_test model_test_test.cpp)
target_include_directories(model_test_test PRIVATE ../../main)
add_test(NAME model_test_sequence COMMAND model_test_test)
add_executable(sd_camera_test sd_camera_test.cpp)
target_include_directories(sd_camera_test PRIVATE ../../main)
target_compile_options(sd_camera_test PRIVATE -UNDEBUG)
add_test(NAME sd_camera_blending COMMAND sd_camera_test)
add_executable(display_buffer_pool_test display_buffer_pool_test.cpp)
target_compile_options(display_buffer_pool_test PRIVATE -UNDEBUG)
add_test(NAME display_buffer_retirement COMMAND display_buffer_pool_test)
add_executable(display_damage_test display_damage_test.cpp)
target_include_directories(display_damage_test PRIVATE ../../main)
target_compile_options(display_damage_test PRIVATE -UNDEBUG)
add_test(NAME display_damage_retirement COMMAND display_damage_test)
add_executable(display_canvas_test display_canvas_test.cpp)
target_link_libraries(display_canvas_test PRIVATE lvgl)
target_compile_options(display_canvas_test PRIVATE -UNDEBUG)
add_test(NAME display_landscape_stride COMMAND display_canvas_test)
add_executable(link_policy_test link_policy_test.cpp)
target_include_directories(link_policy_test PRIVATE ../../main)
target_compile_options(link_policy_test PRIVATE -UNDEBUG)
add_test(NAME usb_bluetooth_priority COMMAND link_policy_test)
if(PET_BUILD_DESKTOP)
    add_library(pet_desktop_ui SHARED ../../desktop/native.cpp ${PET_UI_SOURCES})
    target_compile_definitions(pet_desktop_ui PRIVATE PET_DESKTOP_PREVIEW=1)
    target_include_directories(pet_desktop_ui PRIVATE ../../main)
    target_link_libraries(pet_desktop_ui PRIVATE lvgl m)
endif()
